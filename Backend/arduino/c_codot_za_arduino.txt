#include <EEPROM.h>

#define EEPROM_pH7  0
#define EEPROM_pH4  4
#define EEPROM_FLAG 8
#define CAL_FLAG    0xAB

#define numReadings 15

float temp = 25.0;
float cal_voltage_pH7 = 1.69;
float cal_voltage_pH4 = 0.00;
float slope, intercept;
boolean paused = false;

unsigned long lastRead = 0;
const unsigned long INTERVAL_MS = 1000;

void setup() {
  Serial.begin(9600);

  // Citanje na EEPROM
  if (EEPROM.read(EEPROM_FLAG) == CAL_FLAG) {
    EEPROM.get(EEPROM_pH7, cal_voltage_pH7);
    EEPROM.get(EEPROM_pH4, cal_voltage_pH4);
    slope     = (4.0 - 7.0) / (cal_voltage_pH4 - cal_voltage_pH7);
    intercept = 7.0 - (slope * cal_voltage_pH7);
    Serial.println("Calibration loaded.");
  } else {
    Serial.println("No calibration found, using defaults.");
    slope     = -5.7;
    intercept = 7.0 - (-5.7 * cal_voltage_pH7);
  }

  Serial.println("Send 'P' to pause/resume measurements.");
  Serial.println("pH,PPM,Conductivity_uS_cm");
}

void loop() {

  if (Serial.available() > 0) {
    char cmd = Serial.read();
    while (Serial.available()) Serial.read();

  // Komanda za pauziranje
    if (cmd == 'p' || cmd == 'P') {
      paused = !paused;
      if (paused) {
        Serial.println(">>> PAUSED - send P to resume <<<");
      } else {
        Serial.println(">>> RESUMED <<<");
      }
    }
  }

  if (paused) return;

  // pH merenje i citanje
  float pHVoltageSum = 0;
  for (int i = 0; i < numReadings; i++) {
    int raw = analogRead(A1);
    float voltage = raw * 5.0 / 1024.0;
    pHVoltageSum += voltage;
    delay(20);
  }
  float avgPHVoltage = pHVoltageSum / numReadings;
  float pH = slope * avgPHVoltage + intercept;

  // TDS merenje 
  int readingsTDS[numReadings];
  float tdsVoltageSum = 0;
  for (int i = 0; i < numReadings; i++) {
    readingsTDS[i] = analogRead(A0);
    float tdsVoltage = readingsTDS[i] * 5.0 / 1024.0;
    tdsVoltageSum += tdsVoltage;
    delay(20);
  }
  float tdsVoltage = tdsVoltageSum / numReadings;
  float ppm = (133.42 * pow(tdsVoltage, 3) - 255.86 * pow(tdsVoltage, 2) + 857.39 * tdsVoltage) * 0.5;
  float tempCoef = 1.0 + 0.02 * (temp - 25.0);
  float conductivity = ppm * tempCoef;

  // Zamena za delay na kraj
  unsigned long now = millis();
  if (now - lastRead >= INTERVAL_MS) {
    lastRead = now;

    Serial.print("pH: ");
    Serial.println(pH, 2);
    Serial.print("PPM: ");
    Serial.println(ppm, 2);
    Serial.print("Conductivity [uS/cm]: ");
    Serial.println(conductivity, 2);
    Serial.println();
  }
}
