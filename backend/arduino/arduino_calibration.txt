#include <EEPROM.h>
#define SensorPin A1
#define LED 13
#define samplingInterval 20
#define printInterval 800
#define ArrayLenth  40
#define EEPROM_pH7     0   
#define EEPROM_pH4     4   
#define EEPROM_FLAG    8   
#define CAL_FLAG       0xAB
int pHArray[ArrayLenth];
int pHArrayIndex = 0;
boolean paused = false;
float cal_voltage_pH7 = 1.69;
float cal_voltage_pH4 = 0.00;
int calStep = 0;

void saveCalibration() {
  EEPROM.put(EEPROM_pH7, cal_voltage_pH7);
  EEPROM.put(EEPROM_pH4, cal_voltage_pH4);
  EEPROM.write(EEPROM_FLAG, CAL_FLAG);
  Serial.println(">>> Calibration saved to EEPROM <<<");
}

void loadCalibration() {
  if (EEPROM.read(EEPROM_FLAG) == CAL_FLAG) {
    EEPROM.get(EEPROM_pH7, cal_voltage_pH7);
    EEPROM.get(EEPROM_pH4, cal_voltage_pH4);
    Serial.println(">>> Calibration loaded from EEPROM <<<");
    Serial.print("  pH7 voltage: "); Serial.print(cal_voltage_pH7, 3); Serial.println("V");
    Serial.print("  pH4 voltage: "); Serial.print(cal_voltage_pH4, 3); Serial.println("V");
  } else {
    Serial.println(">>> No saved calibration found, using defaults <<<");
  }
}

void clearCalibration() {
  EEPROM.write(EEPROM_FLAG, 0x00);
  cal_voltage_pH4 = 0.00;
  cal_voltage_pH7 = 1.69;
  Serial.println(">>> Calibration cleared <<<");
}

void setup(void)
{
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
  loadCalibration(); 
  Serial.println("pH meter");
  Serial.println("Send 'P' to pause/resume measurement");
  Serial.println("Send 'C' to calibrate (pH 7, then pH 4)");
}

void loop(void)
{
  static unsigned long samplingTime = millis();
  static unsigned long printTime = millis();
  static float pHValue, voltage;

  if (Serial.available() > 0) {
    char cmd = Serial.read();
    while (Serial.available()) Serial.read();

    if (cmd == 'p' || cmd == 'P') {
      paused = !paused;
      if (paused) {
        memset(pHArray, 0, sizeof(pHArray));
        pHArrayIndex = 0;
        Serial.println(">>> PAUSED - rinse probe, place in next solution, send P to resume <<<");
      } else {
        Serial.println(">>> RESUMED - stabilizing... <<<");
      }
    }
    else if (cmd == 'c' || cmd == 'C') {
      if (calStep == 0) {
        Serial.println(">>> Place probe in pH 7.0 buffer and press C again <<<");
        calStep = 1;
      }
      else if (calStep == 1) {
        Serial.println("Calibrating pH 7...");
        float sum = 0;
        for (int i = 0; i < 20; i++) {
          sum += analogRead(SensorPin);
          delay(50);
        }
        cal_voltage_pH7 = (sum / 20.0) * 5.0 / 1024.0;
        Serial.print("pH 7 voltage set to: ");
        Serial.print(cal_voltage_pH7, 3);
        Serial.println("V");
        Serial.println(">>> Place probe in pH 4.0 buffer and press C again <<<");
        calStep = 2;
      }
      else if (calStep == 2) {
        Serial.println("Calibrating pH 4...");
        float sum = 0;
        for (int i = 0; i < 20; i++) {
          sum += analogRead(SensorPin);
          delay(50);
        }
        cal_voltage_pH4 = (sum / 20.0) * 5.0 / 1024.0;
        Serial.print("pH 4 voltage set to: ");
        Serial.print(cal_voltage_pH4, 3);
        Serial.println("V");

        float slope = (4.0 - 7.0) / (cal_voltage_pH4 - cal_voltage_pH7);
        float intercept = 7.0 - (slope * cal_voltage_pH7);
        Serial.println("=== CALIBRATION COMPLETE ===");
        Serial.print("Use this formula: pH = ");
        Serial.print(slope, 3);
        Serial.print(" * voltage + ");
        Serial.println(intercept, 3);

        saveCalibration(); 

        calStep = 0;
        Serial.println("Calibration done! Press C again to recalibrate");
      }
    }
  }

  if (paused) return;

  if (millis() - samplingTime > samplingInterval)
  {
    pHArray[pHArrayIndex++] = analogRead(SensorPin);
    if (pHArrayIndex == ArrayLenth) pHArrayIndex = 0;
    voltage = avergearray(pHArray, ArrayLenth) * 5.0 / 1024;

    if (cal_voltage_pH4 != 0.00) {
      float slope = (4.0 - 7.0) / (cal_voltage_pH4 - cal_voltage_pH7);
      float intercept = 7.0 - (slope * cal_voltage_pH7);
      pHValue = slope * voltage + intercept;
    } else {
      float intercept = 7.0 - (-5.7 * cal_voltage_pH7);
      pHValue = -5.7 * voltage + intercept;
    }

    samplingTime = millis();
  }

  if (millis() - printTime > printInterval)
  {
    Serial.print("Voltage:");
    Serial.print(voltage, 2);
    Serial.print("    pH value: ");
    Serial.println(pHValue, 2);
    digitalWrite(LED, digitalRead(LED) ^ 1);
    printTime = millis();
  }
}

double avergearray(int* arr, int number) {
  int i;
  int max, min;
  double avg;
  long amount = 0;
  if (number <= 0) {
    Serial.println("Error number for the array to avraging!/n");
    return 0;
  }
  if (number < 5) {
    for (i = 0; i < number; i++) {
      amount += arr[i];
    }
    avg = amount / number;
    return avg;
  } else {
    if (arr[0] < arr[1]) {
      min = arr[0]; max = arr[1];
    }
    else {
      min = arr[1]; max = arr[0];
    }
    for (i = 2; i < number; i++) {
      if (arr[i] < min) {
        amount += min;
        min = arr[i];
      } else {
        if (arr[i] > max) {
          amount += max;
          max = arr[i]; 
        } else {
          amount += arr[i];
        }
      }
    }
    avg = (double)amount / (number - 2);
  }
  return avg;
}
